\subsection{🧹 Tidying: Why It Matters}\label{tidying-why-it-matters}

\begin{quote}
\mbox{}%
\paragraph{❓ What happens if you skip
tidying?}\label{what-happens-if-you-skip-tidying}

\begin{itemize}
\tightlist
\item
  Run regression with the \textbf{\$857K outlier} → experience
  coefficient inflates by \textasciitilde\$300/year
\item
  Don't remove \textbf{duplicates} → p-values look more significant than
  they are (double-counted)
\item
  Leave \textbf{bad column names} → 5-minute debugging session on a
  KeyError typo
\end{itemize}

\textbf{Tidying is not busywork. Tidying is the difference between a
finding and a mistake.}
\end{quote}
